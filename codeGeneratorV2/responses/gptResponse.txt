CREATE COMPUTE MODULE prepareNoContentResponse_Compute
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		SET OutputRoot.Properties = InputRoot.Properties;
		SET OutputLocalEnvironment = InputLocalEnvironment;

		SET OutputLocalEnvironment.Destination.HTTP.ReplyStatusCode = 204;
		SET OutputLocalEnvironment.Variables.TCSLog.context = Environment.Variables.TCSLog.context;
		SET OutputLocalEnvironment.Environment = Environment;

		RETURN TRUE;
	END;

	CREATE PROCEDURE CopyMessageHeaders() BEGIN
		DECLARE I INTEGER 1;
		DECLARE J INTEGER;
		SET J = CARDINALITY(InputRoot.*[]);
		WHILE I < J DO
			SET OutputRoot.*[I] = InputRoot.*[I];
			SET I = I + 1;
		END WHILE;
	END;

	CREATE PROCEDURE CopyEntireMessage() BEGIN
		SET OutputRoot = InputRoot;
	END;
END MODULE;

PATH Shlib_PoliciesReader;

CREATE FILTER MODULE create_case_IsAssys_Filter
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		DECLARE sfcasActivated CHARACTER NormalizeValue(gestionPolicy('{services-integration-policies-credentials-sfcas}:SfcasCredentialsPolicy', 'sfcas-activated'));
		DECLARE assysActivated CHARACTER NormalizeValue(gestionPolicy('{services-integration-policies-credentials-sfcas}:SfcasCredentialsPolicy', 'assys-activated'));
		DECLARE originCallerCode CHARACTER NormalizeValue(Environment.Variables."X-Origin-Caller-Code");
		DECLARE consumerName CHARACTER '';
		DECLARE consumerProcess CHARACTER '';

		CALL ResetRoutingContext();
		CALL ReadCaseContext(consumerName, consumerProcess);

		IF sfcasActivated = 'FALSE' THEN
			CALL ApplyDecision('ASSYS', 'POST_CASE_RULE_1', 'Redirect to Assys', 'sfcas-activated is FALSE.', consumerName, consumerProcess, FALSE);
		ELSEIF assysActivated = 'FALSE' THEN
			CALL ApplyDecision('SFCAS', 'POST_CASE_RULE_2', 'Redirect to SFCAS', 'assys-activated is FALSE.', consumerName, consumerProcess, FALSE);
		ELSEIF originCallerCode = 'SPARX' THEN
			CALL ApplyDecision('EXIT', 'POST_CASE_RULE_3', 'Exit process', 'SPARX cases are handled by Boomi.', consumerName, consumerProcess, FALSE);
		ELSEIF consumerName = 'DIRCALL' AND consumerProcess = 'RSATCS' THEN
			CALL ApplyDecision('SFCAS', 'POST_CASE_RULE_4', 'Redirect to SFCAS', 'DIRCALL and RSATCS detected in ContextList.', consumerName, consumerProcess, FALSE);
		ELSEIF consumerName = 'DIRCALL' AND consumerProcess = 'RSAB2B' THEN
			CALL ApplyDecision('ASSYS', 'POST_CASE_RULE_5', 'Redirect to Assys', 'DIRCALL and RSAB2B detected in ContextList.', consumerName, consumerProcess, FALSE);
		ELSEIF consumerName = 'ONLINEASSY' THEN
			CALL ApplyDecision('SFCAS', 'POST_CASE_RULE_6', 'Redirect to SFCAS', 'ONLINEASSY detected in ContextList.', consumerName, consumerProcess, FALSE);
		ELSEIF IsAssysConsumerName(consumerName) THEN
			CALL ApplyDecision('ASSYS', 'POST_CASE_RULE_7', 'Redirect to Assys', 'B2B / insurance consumer detected in ContextList.', consumerName, consumerProcess, FALSE);
		ELSEIF consumerName = 'MLOG' THEN
			CALL ApplyDecision('ASSYS', 'POST_CASE_RULE_8', 'Redirect to Assys', 'MLOG detected in ContextList.', consumerName, consumerProcess, FALSE);
		ELSE
			CALL ApplyDecision('ASSYS', 'POST_CASE_RULE_13', 'Redirect to Assys', 'Default routing to Assys.', consumerName, consumerProcess, FALSE);
		END IF;

		RETURN Environment.Variables.RoutingDecision = 'ASSYS';
	END;

	CREATE PROCEDURE ResetRoutingContext()
	BEGIN
		DELETE FIELD Environment.Variables.RoutingDecision;
		DELETE FIELD Environment.Variables.TCSLog.context."routing-endpoint";
		DELETE FIELD Environment.Variables.TCSLog.context."routing-rule";
		DELETE FIELD Environment.Variables.TCSLog.context."routing-action";
		DELETE FIELD Environment.Variables.TCSLog.context."routing-detail";
		DELETE FIELD Environment.Variables.TCSLog.context."routing-target";
		DELETE FIELD Environment.Variables.TCSLog.context."routing-consumer-name";
		DELETE FIELD Environment.Variables.TCSLog.context."routing-consumer-process";
		DELETE FIELD Environment.Variables.TCSLog.context."error-code";
		DELETE FIELD Environment.Variables.TCSLog.context."error-message";
		SET Environment.Variables.TCSLog.context.severity = 'INFO';
	END;

	CREATE PROCEDURE ReadCaseContext(INOUT consumerName CHARACTER, INOUT consumerProcess CHARACTER)
	BEGIN
		IF EXISTS(Root.JSON.Data."case".ContextList.Item[1]) THEN
			FOR contextItem AS Root.JSON.Data."case".ContextList.Item[] DO
				CALL ExtractContextValue(contextItem, consumerName, consumerProcess);
			END FOR;
		ELSEIF EXISTS(Root.JSON.Data.ContextList.Item[1]) THEN
			FOR contextItem AS Root.JSON.Data.ContextList.Item[] DO
				CALL ExtractContextValue(contextItem, consumerName, consumerProcess);
			END FOR;
		END IF;
	END;

	CREATE PROCEDURE ExtractContextValue(IN contextItem REFERENCE, INOUT consumerName CHARACTER, INOUT consumerProcess CHARACTER)
	BEGIN
		DECLARE contextKey CHARACTER NormalizeValue(contextItem.ContextElementTypeKey);
		DECLARE contextValue CHARACTER NormalizeValue(COALESCE(contextItem.ContextElementValue, contextItem.ContextElement, ''));

		IF contextKey = 'X-CONSUMER-NAME' AND consumerName = '' THEN
			SET consumerName = contextValue;
		ELSEIF contextKey = 'X-CONSUMER-PROCESS' AND consumerProcess = '' THEN
			SET consumerProcess = contextValue;
		END IF;
	END;

	CREATE PROCEDURE ApplyDecision(
		IN decision CHARACTER,
		IN ruleId CHARACTER,
		IN action CHARACTER,
		IN detail CHARACTER,
		IN consumerName CHARACTER,
		IN consumerProcess CHARACTER,
		IN isError BOOLEAN)
	BEGIN
		SET Environment.Variables.RoutingDecision = decision;
		SET Environment.Variables.TCSLog.context."routing-endpoint" = 'POST_CASE';
		SET Environment.Variables.TCSLog.context."routing-rule" = ruleId;
		SET Environment.Variables.TCSLog.context."routing-action" = action;
		SET Environment.Variables.TCSLog.context."routing-detail" = detail;
		SET Environment.Variables.TCSLog.context."routing-target" = decision;

		IF consumerName <> '' THEN
			SET Environment.Variables.TCSLog.context."routing-consumer-name" = consumerName;
		END IF;

		IF consumerProcess <> '' THEN
			SET Environment.Variables.TCSLog.context."routing-consumer-process" = consumerProcess;
		END IF;

		IF isError THEN
			SET Environment.Variables.TCSLog.context.severity = 'ERROR';
			SET Environment.Variables.TCSLog.context."error-code" = 'ROUTING_ERROR';
			SET Environment.Variables.TCSLog.context."error-message" = detail;
		ELSE
			SET Environment.Variables.TCSLog.context.severity = 'INFO';
			DELETE FIELD Environment.Variables.TCSLog.context."error-code";
			DELETE FIELD Environment.Variables.TCSLog.context."error-message";
		END IF;
	END;

	CREATE FUNCTION IsAssysConsumerName(IN consumerName CHARACTER) RETURNS BOOLEAN
	BEGIN
		RETURN consumerName = 'MOBILITY'
			OR consumerName = 'ARVALDIGI'
			OR consumerName = 'BALOISEDIG'
			OR consumerName = 'BMWDIG'
			OR consumerName = 'DACIADIG'
			OR consumerName = 'FORDDIG'
			OR consumerName = 'KAPO'
			OR consumerName = 'LEXUSDIG'
			OR consumerName = 'MAZDA DIG'
			OR consumerName = 'RENAULTDIG'
			OR consumerName = 'TASDIG'
			OR consumerName = 'TCSCH'
			OR consumerName = 'TOYOTADIG'
			OR consumerName = 'VOLVODIG';
	END;

	CREATE FUNCTION NormalizeValue(IN inputValue CHARACTER) RETURNS CHARACTER
	BEGIN
		RETURN UPPER(TRIM(COALESCE(inputValue, '')));
	END;

END MODULE;

CREATE FILTER MODULE createUpdate_service_IsAssys_Filter
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		DECLARE sfcasActivated CHARACTER NormalizeValue(gestionPolicy('{services-integration-policies-credentials-sfcas}:SfcasCredentialsPolicy', 'sfcas-activated'));
		DECLARE assysActivated CHARACTER NormalizeValue(gestionPolicy('{services-integration-policies-credentials-sfcas}:SfcasCredentialsPolicy', 'assys-activated'));
		DECLARE originCallerCode CHARACTER NormalizeValue(Environment.Variables."X-Origin-Caller-Code");
		DECLARE caseId CHARACTER TRIM(COALESCE(LocalEnvironment.REST.Input.Parameters."case-id", ''));
		DECLARE isReferenceCaseId BOOLEAN common.checkIdReferenceCaseFormat(caseId);

		CALL ResetRoutingContext();

		IF sfcasActivated = 'FALSE' THEN
			CALL ApplyDecision('ASSYS', 'SERVICE_RULE_1', 'Redirect to Assys', 'sfcas-activated is FALSE.', caseId, FALSE);
		ELSEIF assysActivated = 'FALSE' THEN
			CALL ApplyDecision('SFCAS', 'SERVICE_RULE_2', 'Redirect to SFCAS', 'assys-activated is FALSE.', caseId, FALSE);
		ELSEIF originCallerCode = 'SPARX' AND isReferenceCaseId THEN
			CALL ApplyDecision('EXIT', 'SERVICE_RULE_9', 'Exit process', 'SPARX 9 digits case-id is handled by Boomi.', caseId, FALSE);
		ELSEIF originCallerCode = 'DRAPP' AND isReferenceCaseId THEN
			CALL ApplyDecision('ERROR_EXIT', 'SERVICE_RULE_10', 'Log an error and exit process', 'DRAPP 9 digits case-id must use Boomi to communicate with Assys.', caseId, TRUE);
		ELSEIF isReferenceCaseId THEN
			CALL ApplyDecision('ASSYS', 'SERVICE_RULE_11', 'Redirect to Assys', 'case-id is a 9 digits reference.', caseId, FALSE);
		ELSEIF NOT isReferenceCaseId THEN
			CALL ApplyDecision('SFCAS', 'SERVICE_RULE_12', 'Redirect to SFCAS', 'case-id is not a 9 digits reference.', caseId, FALSE);
		ELSE
			CALL ApplyDecision('ASSYS', 'SERVICE_RULE_13', 'Redirect to Assys', 'Default routing to Assys.', caseId, FALSE);
		END IF;

		RETURN Environment.Variables.RoutingDecision = 'ASSYS';
	END;

		DELETE FIELD Environment.Variables.TCSLog.context."routing-case-id";
		DELETE FIELD Environment.Variables.TCSLog.context."error-code";
		DELETE FIELD Environment.Variables.TCSLog.context."error-message";
		SET Environment.Variables.TCSLog.context.severity = 'INFO';
	END;

	CREATE PROCEDURE ApplyDecision(
		IN decision CHARACTER,
		IN ruleId CHARACTER,
		IN action CHARACTER,
		IN detail CHARACTER,
		IN caseId CHARACTER,
		IN isError BOOLEAN)
	BEGIN
		SET Environment.Variables.RoutingDecision = decision;
		SET Environment.Variables.TCSLog.context."routing-endpoint" = 'POST_OR_PUT_SERVICE';
		SET Environment.Variables.TCSLog.context."routing-rule" = ruleId;
		SET Environment.Variables.TCSLog.context."routing-action" = action;
		SET Environment.Variables.TCSLog.context."routing-detail" = detail;
		SET Environment.Variables.TCSLog.context."routing-target" = decision;
		SET Environment.Variables.TCSLog.context."routing-case-id" = caseId;

		IF isError THEN
			SET Environment.Variables.TCSLog.context.severity = 'ERROR';
			SET Environment.Variables.TCSLog.context."error-code" = 'ROUTING_ERROR';
			SET Environment.Variables.TCSLog.context."error-message" = detail;
		ELSE
			SET Environment.Variables.TCSLog.context.severity = 'INFO';
			DELETE FIELD Environment.Variables.TCSLog.context."error-code";
			DELETE FIELD Environment.Variables.TCSLog.context."error-message";
		END IF;
	END;

	CREATE FUNCTION NormalizeValue(IN inputValue CHARACTER) RETURNS CHARACTER

CREATE FILTER MODULE MessageFilter_FilterFSMMessage
	-- Return TRUE if the request must use the FSM callable path (blank or MLOG consumer name).
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		DECLARE caseContextRef REFERENCE TO Root.JSON.Data."case".ContextList;
		DECLARE contextRef REFERENCE TO Root.JSON.Data.ContextList;

		IF CheckContextList(caseContextRef) THEN
			RETURN TRUE;
		END IF;

		RETURN CheckContextList(contextRef);
	END;

	CREATE FUNCTION CheckContextList(IN contextListRef REFERENCE) RETURNS BOOLEAN
	BEGIN
		IF EXISTS(contextListRef.Item[1]) THEN
			FOR contextElement AS contextListRef.Item[] DO
				IF NormalizeValue(contextElement.ContextElementTypeKey) = 'X-CONSUMER-NAME'
					AND (NormalizeValue(COALESCE(contextElement.ContextElementValue, contextElement.ContextElement, '')) = ''
						OR NormalizeValue(COALESCE(contextElement.ContextElementValue, contextElement.ContextElement, '')) = 'MLOG') THEN
					RETURN TRUE;
				END IF;
			END FOR;
		END IF;

		RETURN FALSE;
	END;

	CREATE FUNCTION NormalizeValue(IN inputValue CHARACTER) RETURNS CHARACTER
	BEGIN
		RETURN UPPER(TRIM(COALESCE(inputValue, '')));
	END;

END MODULE;

Changed:
- Added new shared compute module:
  - `common/prepareNoContentResponse_Compute.esql`
  - Prepares an empty HTTP response with status `204`.
- Updated routing/filter ESQL:
  - `create_case_IsAssys_Filter.esql`
  - `createUpdate_service_IsAssys_Filter.esql`
  - `update_case_IsAssys_Filter.esql`
  - `MessageFilter_FilterFSMMessage.esql`
- Updated subflows to support “exit process / log error and exit process” without invoking a backend:
  - `create_case.subflow`
  - `create_service.subflow`
  - `update_service.subflow`
  - `update_case.subflow`
- Updated policy project to add and populate `assys-activated` and to align `sfcas-activated` values:
  - `SfcasCredentialsPolicy.policyxml`
  - `properties/DEV.properties`
  - `properties/QA.properties`
  - `properties/ACP.properties`
  - `properties/PRD.properties`

Behavior implemented:
- Rule order is evaluated with `IF / ELSEIF` so processing stops at the first true rule.
- All string comparisons are case-insensitive via `UPPER(TRIM(...))`.
- Routing decisions are logged into `Environment.Variables.TCSLog.context`:
  - routing-endpoint
  - routing-rule
  - routing-action
  - routing-detail
  - routing-target
  - routing-case-id / routing-consumer-* where relevant
- Post Case supports:
  - policy-based Assys/SFCAS redirection
  - SPARX exit
  - ContextList rules for DIRCALL / RSATCS, DIRCALL / RSAB2B, ONLINEASSY, B2B consumer list, MLOG
  - default Assys
- Put Case / Post Service / Put Service supports:
  - policy-based Assys/SFCAS redirection
  - SPARX + 9-digit case-id exit
  - DRAPP + 9-digit case-id logged error exit
  - 9-digit case-id => Assys
  - non-9-digit case-id => SFCAS
  - default Assys

Assumptions / limitations:
- I assumed the DRAPP rule row in the story contains a typo and should use `DRAPP` instead of `SPARX`. I implemented `DRAPP`.
- “exit process” and “log an error and exit process” were ambiguous from an HTTP contract perspective. I implemented both as an immediate `204 No Content` response and, for the DRAPP case, logged it with `severity = ERROR` in the monitoring context.
- For Post Case, I preserved the existing FSM-specific Assys path selection by keeping `MessageFilter_FilterFSMMessage`; I only made it robust for both `$.case.ContextList[]` and the older `$.ContextList[]` shape.
- I did not change the generated REST msgflow or OpenAPI descriptor because the operation structure remains unchanged.
