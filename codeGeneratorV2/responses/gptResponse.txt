--FN-- insurance-claims-integration-claims/apps/insurance-claims-integration-claims/Routing_ValidatePayloadAndDetermineDestination.esql --FN--
--FC--
CREATE COMPUTE MODULE Routing_ValidatePayloadAndDetermineDestination
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN    
		SET OutputRoot.JSON.Data = InputRoot.JSON.Data;
		SET OutputLocalEnvironment = InputLocalEnvironment;
		
		DELETE FIELD Environment.Variables.CustomValidationMessage;
		SET Environment.Variables.HttpReturnCode = 400;
		
		--MANDATORY FIELDS IN ALL CASES	    
		--X-Caller-Code	
		IF(InputRoot.HTTPInputHeader."X-Origin-Caller-Code" IS NULL OR TRIM(InputRoot.HTTPInputHeader."X-Origin-Caller-Code") = '') THEN
			THROW USER EXCEPTION VALUES('The header X-Origin-Caller-Code is missing or is empty');
			RETURN FALSE;
	    END IF;
	    --Verifie les product-code from attributes if exists --> get from attributes
	    DECLARE productCode ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'product-code');
	    IF EXISTS(productCode.[]) AND productCode."value-code" IS NOT NULL AND TRIM(productCode."value-code") <> '' THEN
	    	SET Environment.Variables.productCode = productCode."value-code";
	    	--policy-number and we get the product-code from it
	    ELSEIF InputRoot.JSON.Data."policy-number" IS NULL OR TRIM(InputRoot.JSON.Data."policy-number") = '' OR STARTSWITH(UPPER(InputRoot.JSON.Data."policy-number"), 'PO') = FALSE THEN	
			THROW USER EXCEPTION VALUES('The input parameter policy-number is missing or null.');
			RETURN FALSE;
		ELSE
			SET Environment.Variables.productCode = SUBSTRING(UPPER(InputRoot.JSON.Data."policy-number") FROM 13 FOR 3);
		END IF;
		--event/date
		IF InputRoot.JSON.Data.event.date IS NULL OR TRIM(InputRoot.JSON.Data.event.date) = '' THEN
			THROW USER EXCEPTION VALUES('The input parameter event/date is missing or null.');
			RETURN FALSE;
		END IF;
		--policyHolder/role and policyHolder/personal-id
		FOR claimContactsRef as InputRoot.JSON.Data."claim-contacts".Item[] DO			
			IF(UPPER(claimContactsRef."role-desc") = UPPER('Policy Holder') OR UPPER(claimContactsRef."role") = UPPER('polHold') OR UPPER(claimContactsRef."role") = UPPER('Policyholder')) THEN
				SET Environment.Variables.claim."claim-contacts".policyHolder.role = claimContactsRef.role; 			
				SET Environment.Variables.claim."claim-contacts".policyHolder."personal-id" = claimContactsRef."personal-id";
		    END IF;			
		END FOR;	
		IF Environment.Variables.claim."claim-contacts".policyHolder.role IS NULL OR TRIM(Environment.Variables.claim."claim-contacts".policyHolder.role) = ''  THEN			
			THROW USER EXCEPTION VALUES('The input parameter claim-contacts/role polHold is missing or null.');
			RETURN FALSE;	
		END IF;
		SET Environment.Variables.TCSLog.context.transaction.POLICYHOLDER_PERSONAL_ID = Environment.Variables.claim."claim-contacts".policyHolder."personal-id";
		DECLARE informant ROW THE(SELECT C FROM InputRoot.JSON.Data."claim-contacts".Item[] AS C WHERE C.role = 'informant');
		IF EXISTS(informant.[]) THEN
			SET Environment.Variables.TCSLog.context.transaction.INFORMANT_PERSONAL_ID = informant."personal-id";
		END IF;
		DECLARE consumerProcess ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'X-CONSUMER-PROCESS');
		DECLARE isLpClaim BOOLEAN UPPER(COALESCE(consumerProcess."value-code",'')) = 'LP_CLAIM';
		IF (Environment.Variables.claim."claim-contacts".policyHolder."personal-id" IS NULL OR TRIM(Environment.Variables.claim."claim-contacts".policyHolder."personal-id") = '') AND isLpClaim IS FALSE THEN
			THROW USER EXCEPTION VALUES('The input parameter claim-contacts/personal-id of the policy Holder is missing or null.');
			RETURN FALSE;	
		END IF;
		
		-- RSA / PA validation for tcs.ch only
		DECLARE isTcsCh BOOLEAN UCASE(COALESCE(InputRoot.HTTPInputHeader."X-Origin-Caller-Code", '')) = 'TCSCH';
		DECLARE claimUseCase ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'claim-use-case');
		DECLARE claimUseCaseValue CHARACTER claimUseCase."value-code";
		DECLARE vehicleCount INTEGER 0;
		FOR claimCoverRef AS InputRoot.JSON.Data."claimed-cover".Item[] DO
			FOR claimedDamagesRef AS claimCoverRef."claimed-damages".Item[] DO
				IF EXISTS(claimedDamagesRef."damaged-objects-details".vehicle.Item[]) THEN
					SET vehicleCount = vehicleCount + CARDINALITY(claimedDamagesRef."damaged-objects-details".vehicle.Item[]);
				END IF;
			END FOR;
		END FOR;
		
		IF isTcsCh THEN
			IF EXISTS(claimUseCase.[]) AND claimUseCaseValue NOT IN ('RSA', 'PA') THEN
				CALL RaiseBadRequest('ACE : claim-use-case value not allowed (expected RSA or PA)');
				RETURN FALSE;
			END IF;
			
			IF claimUseCaseValue = 'RSA' THEN
				IF vehicleCount = 0 THEN
					CALL RaiseBadRequest('ACE : vehicle is mandatory for a RSA claim');
					RETURN FALSE;
				ELSEIF vehicleCount > 1 THEN
					CALL RaiseBadRequest('ACE : only one vehicle is allowed');
					RETURN FALSE;
				END IF;
			ELSEIF vehicleCount > 0 THEN
				CALL RaiseBadRequest('ACE : vehicle is only allowed for a RSA claim');
				RETURN FALSE;
			END IF;
		END IF;
		
		--ROUTING rules to determine the destination of the message:  	
		IF UCASE(InputRoot.HTTPInputHeader."X-Origin-Caller-Code") IN ('LEXFORYOU', 'ASSISTACH') THEN
			SET Environment.Variables.ClaimDestination = 'ALIS';
		ELSE
			SET Environment.Variables.damageCodeId = InputRoot.JSON.Data."claimed-cover".Item[1]."claimed-damages".Item[1]."functional-id";
			IF Environment.Variables.productCode = '103' AND Environment.Variables.damageCodeId NOT IN ('1000043', '1000044', '1000045', '1000046', '1000047',
				'1000048', '1000049', '1000050', '1000051', '1000052', '1000053', '1000054', '1000055') THEN
				SET Environment.Variables.ClaimDestination = 'IDIT';
			ELSEIF Environment.Variables.productCode IN ('101', '102', '105', '106') THEN 
				SET Environment.Variables.ClaimDestination = 'IDIT';
			ELSEIF Environment.Variables.productCode = '115' AND Environment.Variables.damageCodeId NOT IN ('1000126', '1000127', '1000128', '1000129', '1000130') THEN 
				SET Environment.Variables.ClaimDestination = 'IDIT';	
			ELSEIF Environment.Variables.productCode IN ('101', '103', '106')
				AND NOT(EXISTS(Environment.Variables.damageCodeId[]))
				AND EXISTS(InputRoot.JSON.Data.event."primary-cause-of-loss-extcode"[]) THEN 
				SET Environment.Variables.ClaimDestination = 'IDIT';	
			ELSEIF Environment.Variables.productCode IN ('103', '107', '108', '109', '110', '114', '115') THEN	
				SET Environment.Variables.ClaimDestination = 'ALIS';
			ELSE
				THROW USER EXCEPTION VALUES('The product code ' || Environment.Variables.productCode || ' indicated in the payload cannot be processed.');
				RETURN FALSE;	
			END IF;		
		END IF;
		
		
		--MANDATORY FIELDS IN ALL CASES IF THE DESTINATION IS IDIT
		IF Environment.Variables.ClaimDestination = 'IDIT' THEN
		   	--event/description
			IF InputRoot.JSON.Data.event.description IS NULL OR TRIM(InputRoot.JSON.Data.event.description) = '' THEN
				THROW USER EXCEPTION VALUES('The input parameter event/description is missing or null.');
				RETURN FALSE;
			END IF;
			--event/primary-cause-of-loss-extcode
			IF InputRoot.JSON.Data.event."primary-cause-of-loss-extcode" IS NULL OR TRIM(InputRoot.JSON.Data.event."primary-cause-of-loss-extcode") = '' THEN	
				THROW USER EXCEPTION VALUES('The input parameter event/primary-cause-of-loss-extcode is missing or null.');
				RETURN FALSE;
			END IF;
			--event/location/country-code
			IF InputRoot.JSON.Data.event.location."country-code" IS NULL OR TRIM(InputRoot.JSON.Data.event.location."country-code") = '' THEN	
				THROW USER EXCEPTION VALUES('The input parameter event/location/country-code is missing or null.');
				RETURN FALSE;
			END IF;
			--event/location/type-desc
			IF InputRoot.JSON.Data.event.location."type-desc" IS NULL OR TRIM(InputRoot.JSON.Data.event.location."type-desc") = '' THEN
				THROW USER EXCEPTION VALUES('The input parameter event/location/type-desc is missing or null.');
				RETURN FALSE;
			END IF;
			--informant/role and informant/personal-id
			IF NOT EXISTS(informant.[]) OR informant."personal-id" IS NULL OR TRIM(informant."personal-id") = '' THEN
			    THROW USER EXCEPTION VALUES('The input parameter claim-contacts/role or claim-contacts/personal-id of the informant is missing or null.');
			    RETURN FALSE;
			END IF;
		END IF;				
		--MANDATORY FIELDS IF THE DESTINATION IS IDIT AND THE PRODUCT CODE = 103 (ETI)
		IF Environment.Variables.ClaimDestination = 'IDIT' AND Environment.Variables.productCode = '103' THEN
			FOR claimCoverRef AS InputRoot.JSON.Data."claimed-cover".Item[] DO
			    FOR claimedDamagesRef AS claimCoverRef."claimed-damages".Item[] DO
			    	DECLARE claimedDamagesTravelInfos ROW claimedDamagesRef."damaged-objects-details"."travel-infos";			
			    	DECLARE damageCode CHARACTER claimedDamagesRef."functional-id"; 
			    	IF damageCode IN ('1000005', '1000006', '1000007') THEN
			    		--travel-infos/planned-travel-departure-date
						IF claimedDamagesTravelInfos."planned-travel-departure-date" IS NULL OR TRIM(claimedDamagesTravelInfos."planned-travel-departure-date") = '' THEN
							THROW USER EXCEPTION VALUES('The input parameter travel-infos/planned-travel-departure-date is missing or null in one of the claimed damages for the product-code 103
							 						     and damage code ' || damageCode);
							RETURN FALSE;
						END IF;
						--travel-infos/impediment-date	
						IF claimedDamagesTravelInfos."impediment-date" IS NULL OR TRIM(claimedDamagesTravelInfos."impediment-date") = ''  THEN			
							THROW USER EXCEPTION VALUES('The input parameter travel-infos/impediment-date is missing or null in one of the claimed damages for the product-code 103
														 and damage code ' || damageCode);
							RETURN FALSE;	
						END IF;
			    	END IF;		    	
					--IF THE DAMAGE CODE ID = 1000039 (Rental Car franchise payment)
					DECLARE damageCodeId ROW claimedDamagesRef."functional-id";
					IF damageCodeId = '1000039' THEN
						DECLARE claimedDamagesCDW ROW claimedDamagesRef."damaged-objects-details".CDW;		
					    --CDW/vehicle-type-one
						IF claimedDamagesCDW."vehicle-type-one" IS NULL THEN
							THROW USER EXCEPTION VALUES('The input parameter CDW/vehicle-type-one is missing in the claimed damage with the id 1000039 for the product-code 103.');
							RETURN FALSE;
						END IF;	
					END IF;
			    END FOR;
			END FOR;    					
		END IF;	
		--MANDATORY FIELDS IF THE DESTINATION IS IDIT AND THE PRODUCT CODE = 105 (Bike)
		IF Environment.Variables.ClaimDestination = 'IDIT' AND Environment.Variables.productCode = '105' THEN
			FOR claimCoverRef AS InputRoot.JSON.Data."claimed-cover".Item[] DO
			    FOR claimedDamagesRef AS claimCoverRef."claimed-damages".Item[] DO
			    	DECLARE claimedDamagesBike ROW claimedDamagesRef."damaged-objects-details".bike;		
				    --Bike/type
					IF claimedDamagesBike.type IS NULL OR TRIM(claimedDamagesBike.type) = '' THEN	
						THROW USER EXCEPTION VALUES('The input parameter Bike/type is missing or null in one of the claimed damages for the product-code 105.');
						RETURN FALSE;
					END IF;
					--Bike/brand
					IF claimedDamagesBike.brand IS NULL OR TRIM(claimedDamagesBike.brand) = '' THEN
						THROW USER EXCEPTION VALUES('The input parameter Bike/brand is missing or null in one of the claimed damages for the product-code 105.');
						RETURN FALSE;
					END IF;
					--Bike/25km-more = true, Bike/plate
					IF claimedDamagesBike."25km-more" = 'true' AND COALESCE(TRIM(claimedDamagesBike.plate), '') = '' THEN
                    	THROW USER EXCEPTION VALUES('The input parameter bike/25km-more is true, but the input parameter bike/plate is missing or null in one of the claimed damages for the product-code 105.');
                    	RETURN FALSE;
                    END IF;
			    END FOR;
			END FOR;    	
		END IF;
		--MANDATORY ATTRIBUTE FIELDS AND MANDATORY SUBATTRIBUTE FIELDS FOR THE document-list-guuid ATTRIBUTE IF THE DESTINATION IS IDIT
		IF Environment.Variables.ClaimDestination = 'IDIT' THEN
			--total-estimated-damaged-currency-amount & total-estimated-damaged-currency-code 
			IF (EXISTS(InputRoot.JSON.Data."total-estimated-damaged-currency-amount"[]) OR COALESCE(InputRoot.JSON.Data."total-estimated-damaged-currency-code", '') <> '')
			   AND (NOT(EXISTS(InputRoot.JSON.Data."total-estimated-damaged-currency-amount"[])) OR COALESCE(InputRoot.JSON.Data."total-estimated-damaged-currency-code", '') = '') THEN 
			   		THROW USER EXCEPTION VALUES('total-estimated-damaged-currency-amount and total-estimated-damaged-currency-code must be present together');
			    	RETURN FALSE;
			END IF;
			--EXTERNAL-CLAIM
			--external-claim-id
			FOR externalClaim AS InputRoot.JSON.Data."external-claim".Item[] DO
				IF COALESCE(externalClaim."external-claim-system", '') <> '' AND COALESCE(externalClaim."external-claim-id", '') = '' THEN 
					THROW USER EXCEPTION VALUES('external-claim-system is present but external-claim-id is missing');
			    	RETURN FALSE;
				END IF;
			END FOR;
			--ATTRIBUTES
		   	--e-mail	
			DECLARE email ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'e-mail');
			IF NOT EXISTS(email.[]) OR email."value-code" IS NULL OR TRIM(email."value-code") = '' THEN
			    THROW USER EXCEPTION VALUES('The email attribute in the Claim object is missing.');
			    RETURN FALSE;
			END IF;
			--language
			DECLARE language ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'language');
			IF NOT EXISTS(language.[]) OR language."value-code" IS NULL OR NOT language."value-code" IN ('FR', 'DE', 'IT') THEN
			    THROW USER EXCEPTION VALUES('The language attribute in the Claim object is missing or is not one of the accepted values (FR, DE, IT).');
			    RETURN FALSE;
			END IF;
			--personal.reference
			DECLARE personalReference ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'personal-reference');	
			IF NOT EXISTS(personalReference.[]) OR personalReference."value-code" IS NULL OR TRIM(personalReference."value-code") = '' THEN
			    THROW USER EXCEPTION VALUES('The personal-reference attribute in the Claim object is missing.');
			    RETURN FALSE;
			END IF;
			--number-of-documents	
			DECLARE numberOfDocuments ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'number-of-documents');
			IF NOT EXISTS(numberOfDocuments.[]) OR numberOfDocuments."value-code" IS NULL THEN
			    THROW USER EXCEPTION VALUES('The number-of-documents attribute in the Claim object is missing.');
			    RETURN FALSE;
			END IF;
			--document-list-guuid
			DECLARE documentListGUUID ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'document-list-guuid');
		    IF numberOfDocuments."value-code" > 0 AND (NOT EXISTS(documentListGUUID.[]) OR documentListGUUID."value-code" IS NULL OR TRIM(documentListGUUID."value-code") = '') THEN    
		        THROW USER EXCEPTION VALUES('The document-list-guuid attribute in the Claim object is missing with the value of the number-of-documents attribute > 0.');
			    RETURN FALSE;
		    END IF;
			--difference between the value of the field number-of-documents and the number of document sub-attributes
			IF numberOfDocuments."value-code" <> CARDINALITY(documentListGUUID."sub-attributes".Item[]) THEN
				THROW USER EXCEPTION VALUES('There is a difference between the value-code of the number-of-documents attribute and the number of document sub-attributes.');
			    RETURN FALSE;
			END IF;
			--SUBATTRIBUTES
			IF numberOfDocuments."value-code" > 0 THEN   	
				FOR SubAttribute AS documentListGUUID."sub-attributes".Item[] DO
					--document-uuid	
					DECLARE documentUUID ROW THE(SELECT A FROM SubAttribute.Item[] AS A WHERE A.type = 'document-uuid');
					IF NOT EXISTS(documentUUID.[]) OR documentUUID."value-code" IS NULL OR TRIM(documentUUID."value-code") = '' THEN
					    THROW USER EXCEPTION VALUES('The document-uuid sub-attribute is missing for one of the documents in the document-list-guuid attribute.');
					    RETURN FALSE;
					END IF;
					--document-type
					DECLARE documentType ROW THE(SELECT A FROM SubAttribute.Item[] AS A WHERE A.type = 'document-type');
					IF NOT EXISTS(documentType.[]) OR documentType."value-code" IS NULL OR TRIM(documentType."value-code") = '' THEN
					    THROW USER EXCEPTION VALUES('The document-type sub-attribute is missing for one of the documents in the document-list-guuid attribute.');
					    RETURN FALSE;
					END IF;
					--file-name
					DECLARE fileName ROW THE(SELECT A FROM SubAttribute.Item[] AS A WHERE A.type = 'file-name');
					IF NOT EXISTS(fileName.[]) OR fileName."value-code" IS NULL OR TRIM(fileName."value-code") = '' THEN
					    THROW USER EXCEPTION VALUES('The file-name sub-attribute is missing for one of the documents in the document-list-guuid attribute.');
					    RETURN FALSE;
					END IF;	
				END FOR;	
			END IF;
		END IF;
		
		SET Environment.Variables.HttpReturnCode = UNKNOWN;	
				
		RETURN TRUE;	
	END;

	CREATE PROCEDURE RaiseBadRequest(IN message CHARACTER)
	BEGIN
		SET Environment.Variables.HttpReturnCode = 400;
		SET Environment.Variables.CustomValidationMessage = message;
		THROW USER EXCEPTION VALUES(message);
	END;

END MODULE;

CREATE COMPUTE MODULE Routing_DetermineIditOperation
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		SET OutputRoot = InputRoot;
		SET OutputLocalEnvironment = InputLocalEnvironment;
		
		DECLARE claimUseCase ROW THE(SELECT A FROM InputRoot.JSON.Data.attributes.Item[] AS A WHERE A.type = 'claim-use-case');
		DECLARE claimUseCaseValue CHARACTER COALESCE(claimUseCase."value-code", '');
		DECLARE routeUsed CHARACTER '';
		
		IF Environment.Variables.ClaimDestination = 'IDIT' THEN
			IF claimUseCaseValue = 'RSA' THEN
				SET routeUsed = 'manageClaims';
			ELSEIF claimUseCaseValue = 'PA' THEN
				SET routeUsed = 'loss-event';
			ELSE
				-- November 2026 fallback.
				-- Remove this block once claim-use-case becomes the single routing driver.
				IF UCASE(COALESCE(Environment.Variables.XOriginCallerCode, '')) = 'TCSCH'
					AND Environment.Variables.productCode IN ('101', '102', '103', '106') THEN
					SET routeUsed = 'manageClaims';
				ELSE
					SET routeUsed = 'loss-event';
				END IF;
			END IF;
		ELSE
			SET routeUsed = 'not-applicable';
		END IF;
		
		IF claimUseCaseValue = '' THEN
			SET Environment.Variables.claimUseCase = 'missing';
		ELSE
			SET Environment.Variables.claimUseCase = claimUseCaseValue;
		END IF;
		
		SET Environment.Variables.routeUsed = routeUsed;
		
		SET Environment.Variables.TCSLog.context."claim-use-case" = Environment.Variables.claimUseCase;
		SET Environment.Variables.TCSLog.context."route-used" = routeUsed;
		
		SET Environment.Variables.userDefined.claimUseCase = Environment.Variables.claimUseCase;
		SET Environment.Variables.userDefined.routeUsed = routeUsed;
		
		RETURN TRUE;
	END;
END MODULE;

--FN-- insurance-claims-integration-claims/apps/insurance-claims-integration-claims/Routing_BuildError.esql --FN--
--FC--
CREATE COMPUTE MODULE Routing_BuildError
CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		-- parse the exception list tree (if there is one) in order to fill the http response returned afterwards
		IF EXISTS(InputExceptionList.*[]) THEN
			CALL parseExceptionList(InputExceptionList, Environment.Variables.ExceptionNumber, Environment.Variables.ExceptionMessage, Environment.Variables.ExceptionStackTrace);
		END IF;
		if EXISTS(InputRoot.HTTPResponseHeader."X-Original-HTTP-Status-Code"[]) then
			DECLARE jsonAsBlob BLOB;
			if EXISTS (InputRoot.BLOB.BLOB[]) then
				set jsonAsBlob  = InputRoot.BLOB.BLOB;
			elseif EXISTS (InputRoot.JSON.Data[]) then
				set jsonAsBlob  = ASBITSTREAM(InputRoot.JSON.Data CCSID COALESCE(InputRoot.Properties.CodedCharSetId,1208));
			end if;
			DECLARE jsonAsChar CHAR CAST(jsonAsBlob AS CHAR CCSID 1208);
			set Environment.Variables.ExceptionStackTrace = jsonAsChar;
		end if;
		-- check provider, if empty this means that it's necessarily ACE that is causing the error 
		SET Environment.Variables.ErrorProviderName = COALESCE(Environment.Variables.ErrorProviderName, 'ACE');
		
		-- check if the environment contains an http status error code to return (if not by default we return 500) 
		IF Environment.Variables.HttpReturnCode IS NULL OR CARDINALITY(Environment.Variables.HttpReturnCode[]) = 0 THEN
			SET Environment.Variables.HttpReturnCode = COALESCE(InputRoot.HTTPResponseHeader."X-Original-HTTP-Status-Code",500);
			if Environment.Variables.ErrorProviderName = 'APIC' and Environment.Variables.HttpReturnCode = 401 then
				set Environment.Variables.ErrorRefApp = 'ACETECH0011';
			end if;
		END IF;
		-- create the http response (http status & body)
		SET OutputLocalEnvironment.Destination.HTTP.ReplyStatusCode = Environment.Variables.HttpReturnCode;
		SET OutputRoot.JSON.Data.timestamp = CAST(CURRENT_TIMESTAMP AS CHARACTER FORMAT 'I');
		SET OutputRoot.JSON.Data.httpCode = Environment.Variables.ErrorProviderName || ' : ' || CAST(COALESCE(Environment.Variables.ErrorProviderHttpReturnCode, Environment.Variables.HttpReturnCode) AS CHARACTER);
		IF Environment.Variables.HttpReturnCode = 400 THEN
			SET OutputRoot.JSON.Data.httpMessage = Environment.Variables.ErrorProviderName || ' : Bad Request';
			SET OutputRoot.JSON.Data.contexteCode = Environment.Variables.ErrorRefApp;
			IF COALESCE(Environment.Variables.CustomValidationMessage, '') <> '' THEN
				SET OutputRoot.JSON.Data.moreInformation = Environment.Variables.CustomValidationMessage;

CREATE COMPUTE MODULE CreateClaim_mappingClaimToManageClaims
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		CREATE FIELD OutputRoot.JSON.Data;
		
		DECLARE inRef REFERENCE TO InputRoot.JSON.Data;
		DECLARE outRef REFERENCE TO OutputRoot.JSON.Data;
		
		DECLARE policyHolder ROW THE(SELECT C FROM inRef."claim-contacts".Item[] AS C WHERE C.role = 'polHold');
		DECLARE informant ROW THE(SELECT C FROM inRef."claim-contacts".Item[] AS C WHERE C.role = 'informant');
		DECLARE driver ROW THE(SELECT C FROM inRef."claim-contacts".Item[] AS C WHERE C.role = 'driver');
		DECLARE policeIntervention ROW THE(SELECT A FROM inRef.event.attributes.Item[] AS A WHERE A.type = 'police-intervention');
		
		DECLARE vehicleFound BOOLEAN FALSE;
		DECLARE vehicleCategoryExtCode CHARACTER '';
		DECLARE vehicleTypeExtCode CHARACTER '';
		DECLARE vehicleBrandExtCode CHARACTER '';
		DECLARE vehicleModel CHARACTER '';
		DECLARE vehiclePlateRaw CHARACTER '';
		DECLARE vehicleMatriculeNumber CHARACTER '';
		DECLARE vehicleFirstRegistrationDate CHARACTER '';
		DECLARE vehicleMaximumWeight CHARACTER '';
		DECLARE vehicleMaximumHeight CHARACTER '';
		DECLARE vehiclePurchaseDate CHARACTER '';
		DECLARE vehiclePurchaseValue CHARACTER '';
		
		vehicleSearch: BEGIN
			FOR claimCoverRef AS inRef."claimed-cover".Item[] DO
				FOR claimedDamagesRef AS claimCoverRef."claimed-damages".Item[] DO
					FOR vehicleRef AS claimedDamagesRef."damaged-objects-details".vehicle.Item[] DO
						SET vehicleCategoryExtCode = COALESCE(vehicleRef."category-ext-code", '');
						SET vehicleTypeExtCode = COALESCE(vehicleRef."type-ext-code", '');
						SET vehicleBrandExtCode = COALESCE(vehicleRef."brand-ext-code", '');
						SET vehicleModel = COALESCE(vehicleRef.model, '');
						SET vehiclePlateRaw = REPLACE(COALESCE(vehicleRef.plate, ''), ' ', '');
						SET vehicleMatriculeNumber = COALESCE(vehicleRef."matricule-number", '');
						SET vehicleFirstRegistrationDate = COALESCE(vehicleRef."first-reg-date", '');
						SET vehicleMaximumWeight = COALESCE(CAST(vehicleRef."maximum-weight" AS CHARACTER), '');
						SET vehicleMaximumHeight = COALESCE(CAST(vehicleRef."maximum-height" AS CHARACTER), '');
						SET vehiclePurchaseDate = COALESCE(vehicleRef."purchase-date", '');
						SET vehiclePurchaseValue = COALESCE(CAST(vehicleRef."purchase-value" AS CHARACTER), '');
						SET vehicleFound = TRUE;
						LEAVE vehicleSearch;
					END FOR;
				END FOR;
			END FOR;
		END;
		
		SET outRef.messageId = COALESCE(Environment.Variables.XCorrelationID, UUIDASCHAR);
		SET outRef.channelType.extCode = 3;
		
		DECLARE claimIBORef REFERENCE TO outRef;
		CREATE LASTCHILD OF outRef AS claimIBORef TYPE Name NAME 'claimIBO';
		
		SET claimIBORef.externalPolicyNumber = inRef."policy-number";
		SET claimIBORef.claimEventDate = inRef.event.date;
		SET claimIBORef.claimEventDescription = inRef.event.description;
		SET claimIBORef.claimCauseOfLossVO.extCode = inRef.event."primary-cause-of-loss-extcode";
		SET claimIBORef.claimReportingChannel.extCode = 'Web';
		SET claimIBORef.brandCompanyVO.extCode = 'TCS';
		SET claimIBORef.latestChannelOfUpdate.extCode = 'no_update_made';
		
		-- Constants grouped together so they can be removed easily after IRND-16006.
		SET claimIBORef.claimStatus.extCode = 'open';
		
		IF EXISTS(inRef."total-estimated-damaged-currency-amount"[]) THEN
			SET claimIBORef.totalEstimatedDamage = CAST(inRef."total-estimated-damaged-currency-amount" AS DECIMAL);
			SET claimIBORef.totalEstimatedDamageCurrency.extCode = 'CHF';
		END IF;
		
		IF EXISTS(inRef."number-of-beneficiaries"[]) THEN
			SET claimIBORef.numberOfBeneficiaries = CAST(inRef."number-of-beneficiaries" AS INTEGER DEFAULT NULL);
		END IF;
		
		IF EXISTS(policeIntervention.[]) THEN
			IF policeIntervention."value-code" = 'true' THEN
				SET claimIBORef.isPoliceIntervention = TRUE;
			ELSEIF policeIntervention."value-code" = 'false' THEN
				SET claimIBORef.isPoliceIntervention = FALSE;
			END IF;
		END IF;
		
		FOR serviceProviderRef AS inRef."claim-service-providers".Item[] DO
			DECLARE serviceProviderOutRef REFERENCE TO OutputRoot.JSON.Data;
			CREATE LASTCHILD OF claimIBORef.(JSON.Array)serviceProviders AS serviceProviderOutRef NAME 'Item';
			SET serviceProviderOutRef.contactExtNumber = serviceProviderRef."personal-id";
		END FOR;
		
		SET claimIBORef.claimEventLocationIBO.countryVO.extCode = inRef.event.location."country-code";
		SET claimIBORef.claimEventLocationIBO.locationType.extCode = 'country';
		-- TODO: map claimIBO.claimEventLocationIBO.latitude when the source and transformation rules are provided.
		-- TODO: map claimIBO.claimEventLocationIBO.longitude when the source and transformation rules are provided.
		SET claimIBORef.claimEventLocationIBO.addressIBO.cityName = inRef.event.location.city;
		SET claimIBORef.claimEventLocationIBO.addressIBO.zipCode = inRef.event.location."zip-code";
		SET claimIBORef.claimEventLocationIBO.addressIBO.streetName = inRef.event.location."street-name";
		SET claimIBORef.claimEventLocationIBO.addressIBO.houseNr = inRef.event.location."house-number";
		
		IF EXISTS(informant.[]) THEN
			SET claimIBORef.claimInformantIBO.contactExternalNumber = informant."personal-id";
			IF CAST(informant."informant-type" AS CHARACTER) = '1' THEN
				SET claimIBORef.claimInformantIBO.informantType.extCode = 'policyholder';
			ELSEIF COALESCE(CAST(informant."informant-type" AS CHARACTER), '') <> '' THEN
				SET claimIBORef.claimInformantIBO.informantType.extCode = CAST(informant."informant-type" AS CHARACTER);
			END IF;
		END IF;
		
		IF vehicleFound THEN
			DECLARE damagedVehicleRef REFERENCE TO OutputRoot.JSON.Data;
			CREATE LASTCHILD OF claimIBORef.(JSON.Array)damagedVehicleDetails AS damagedVehicleRef NAME 'Item';
			SET damagedVehicleRef.vehicleCategory.extCode = vehicleCategoryExtCode;
			SET damagedVehicleRef.licensePlateNumber = vehiclePlateRaw;
			SET damagedVehicleRef.vehicleType.extCode = vehicleTypeExtCode;
			SET damagedVehicleRef.vehicleBrand.extCode = vehicleBrandExtCode;
			SET damagedVehicleRef.vehicleModel = vehicleModel;
			SET damagedVehicleRef.vehicleMatriculeNumber = vehicleMatriculeNumber;
			SET damagedVehicleRef.firstRegistrationDate = vehicleFirstRegistrationDate;
			SET damagedVehicleRef.maximumWeight = CAST(vehicleMaximumWeight AS INTEGER DEFAULT NULL);
			SET damagedVehicleRef.maximumHeight = CAST(vehicleMaximumHeight AS INTEGER DEFAULT NULL);
			IF vehiclePurchaseDate <> '' THEN
				SET damagedVehicleRef.purchaseDate = vehiclePurchaseDate;
			END IF;
			IF vehiclePurchaseValue <> '' THEN
				SET damagedVehicleRef.purchaseValue = CAST(vehiclePurchaseValue AS DECIMAL DEFAULT NULL);
			END IF;
		END IF;
		
		IF EXISTS(driver.[]) THEN
			SET claimIBORef.driverDetails."DriverContactExtNumber" = driver."personal-id";
			
			DECLARE highAlcohol ROW THE(SELECT A FROM driver.attributes.Item[] AS A WHERE A.type = 'high-alcohol');
			DECLARE alcoholTestResult ROW THE(SELECT A FROM driver.attributes.Item[] AS A WHERE A.type = 'alcohol-test-result');
			DECLARE influenceOfMedicines ROW THE(SELECT A FROM driver.attributes.Item[] AS A WHERE A.type = 'influence-of-medicines');
			DECLARE influenceOfDrugs ROW THE(SELECT A FROM driver.attributes.Item[] AS A WHERE A.type = 'influence-of-drugs');
			DECLARE influenceOfTiredness ROW THE(SELECT A FROM driver.attributes.Item[] AS A WHERE A.type = 'influence-of-tiredness');
			DECLARE driverNotes ROW THE(SELECT A FROM driver.attributes.Item[] AS A WHERE A.type = 'driver-notes');
			
			SET claimIBORef.driverDetails.highAlcohol = ToYesNo(highAlcohol."value-code");
			IF EXISTS(alcoholTestResult.[]) THEN
				SET claimIBORef.driverDetails.alcoholTestResult = CAST(alcoholTestResult."value-code" AS DECIMAL DEFAULT NULL);
			END IF;
			SET claimIBORef.driverDetails.influenceOfMedicines = ToYesNo(influenceOfMedicines."value-code");
			SET claimIBORef.driverDetails.influenceOfDrugs = ToYesNo(influenceOfDrugs."value-code");
			SET claimIBORef.driverDetails.influenceOfTiredness = ToYesNo(influenceOfTiredness."value-code");
			SET claimIBORef.driverDetails.notes = driverNotes."value-code";
		END IF;
		
		DECLARE claimantIBORef REFERENCE TO OutputRoot.JSON.Data;
		CREATE LASTCHILD OF claimIBORef.(JSON.Array)claimantIBOs AS claimantIBORef NAME 'Item';
		SET claimantIBORef.contactExternalNumber = policyHolder."personal-id";
		SET claimantIBORef.claimantSide.extCode = 'policyholder';
		SET claimantIBORef.claimantLiability.extCode = 'unknown';
		-- Constants grouped together so they can be removed easily after IRND-16006.
		SET claimantIBORef.claimantStatus.extCode = 'open';
		
		DECLARE damagedObjectRef REFERENCE TO OutputRoot.JSON.Data;
		CREATE LASTCHILD OF claimantIBORef.(JSON.Array)damagedObjects AS damagedObjectRef NAME 'Item';
		IF vehicleFound THEN
			SET damagedObjectRef.description = BuildDamagedObjectDescription(vehicleModel, vehiclePlateRaw);
		END IF;
		
		FOR claimCoverRef AS inRef."claimed-cover".Item[] DO
			FOR claimedDamagesRef AS claimCoverRef."claimed-damages".Item[] DO
				IF EXISTS(claimedDamagesRef."damaged-objects-details"."damage-items".Item[]) THEN
					FOR damageItemRef AS claimedDamagesRef."damaged-objects-details"."damage-items".Item[] DO
						DECLARE damagedItemOutRef REFERENCE TO OutputRoot.JSON.Data;
						CREATE LASTCHILD OF damagedObjectRef.(JSON.Array)damagedItems AS damagedItemOutRef NAME 'Item';
						SET damagedItemOutRef.itemType.extCode = damageItemRef."ext-code";
						SET damagedItemOutRef.desc = damageItemRef.description;
						IF EXISTS(damageItemRef."purchase-value"[]) THEN
							SET damagedItemOutRef.purchaseValue = CAST(damageItemRef."purchase-value" AS DECIMAL);
						END IF;
						SET damagedItemOutRef.isActive = TRUE;
					END FOR;
				ELSEIF EXISTS(claimedDamagesRef."damaged-objects-details"."damaged-items".Item[]) THEN
					FOR damageItemRef AS claimedDamagesRef."damaged-objects-details"."damaged-items".Item[] DO
						DECLARE damagedItemOutRef REFERENCE TO OutputRoot.JSON.Data;
						CREATE LASTCHILD OF damagedObjectRef.(JSON.Array)damagedItems AS damagedItemOutRef NAME 'Item';
						SET damagedItemOutRef.itemType.extCode = damageItemRef."ext-code";
						SET damagedItemOutRef.desc = damageItemRef.description;
						IF EXISTS(damageItemRef."purchase-value"[]) THEN
							SET damagedItemOutRef.purchaseValue = CAST(damageItemRef."purchase-value" AS DECIMAL);
						END IF;
						SET damagedItemOutRef.isActive = TRUE;
					END FOR;
				END IF;
			END FOR;
		END FOR;
		
		FOR beneficiaryRef AS inRef."claim-contacts".Item[] DO
			IF beneficiaryRef.role = 'beneficiary' THEN
				DECLARE claimContactRef REFERENCE TO OutputRoot.JSON.Data;
				CREATE LASTCHILD OF claimIBORef.(JSON.Array)"ClaimContactRefIBO" AS claimContactRef NAME 'Item';
				SET claimContactRef.contactExternalNumber = beneficiaryRef."personal-id";
				SET claimContactRef.claimContactRoleVO.extCode = 'beneficiary';
			END IF;
		END FOR;
		
		RETURN TRUE;
	END;
	
	CREATE FUNCTION ToYesNo(IN inputValue CHARACTER) RETURNS CHARACTER
	BEGIN
		IF inputValue = 'true' THEN
			RETURN 'Yes';
		ELSEIF inputValue = 'false' THEN
			RETURN 'No';
		END IF;
		RETURN NULL;
	END;
	
	CREATE FUNCTION BuildDamagedObjectDescription(IN model CHARACTER, IN plate CHARACTER) RETURNS CHARACTER
	BEGIN
		DECLARE formattedPlate CHARACTER '';
		
		IF plate IS NOT NULL AND plate <> '' THEN
			IF LENGTH(plate) = 8 THEN
				SET formattedPlate = SUBSTRING(plate FROM 1 FOR 2) || ' ' || SUBSTRING(plate FROM 3 FOR 3) || ' ' || SUBSTRING(plate FROM 6 FOR 3);
			ELSE
				SET formattedPlate = plate;
			END IF;
		END IF;
		
		IF COALESCE(model, '') <> '' AND formattedPlate <> '' THEN
			RETURN model || ' - ' || formattedPlate;
		ELSEIF COALESCE(model, '') <> '' THEN
			RETURN model;
		ELSE
			RETURN formattedPlate;
		END IF;
	END;
END MODULE;
