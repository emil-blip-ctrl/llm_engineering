CREATE COMPUTE MODULE PrepareProcessFile
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		
		-- extract policy values
		DECLARE "ace-host-finance" CHARACTER Shlib_PoliciesReader.gestionPolicy('{common-integration-policies-endpoints}:AceEndpointsPolicy',  'ace-host-finance');
		DECLARE requestPathProperty CHARACTER;
		
		IF COALESCE(Environment.Variables.IngestionSource, '') = 'SANSYS' THEN
			SET requestPathProperty = 'path-finance-accounting-integration-invoices-v2-receivable-documents';
		ELSE
			SET requestPathProperty = 'path-finance-accounting-integration-invoice';
		END IF;
		
		SET Environment.Variables.Policy.RequestURL_invoice = "ace-host-finance" || Shlib_PoliciesReader.gestionPolicy('{finance-integration-policies-endpoints}:AceEndPointsPolicy', requestPathProperty);
		
		CREATE FIELD Environment.Variables.Policy."finance-integration-policies-endpoints".SftpEndPointsPolicy;
		DECLARE RefSftpEndPointsPolicy REFERENCE TO Environment.Variables.Policy."finance-integration-policies-endpoints".SftpEndPointsPolicy;
		call Shlib_PoliciesReader.getAllPolicyProperties('finance-integration-policies-endpoints','SftpEndPointsPolicy',RefSftpEndPointsPolicy);
		if EXISTS(RefSftpEndPointsPolicy.error[]) then
			THROW USER EXCEPTION VALUES('''{finance-integration-policies-endpoints''}:SftpEndPointsPolicy not found'); 
		elseif not EXISTS(RefSftpEndPointsPolicy.post_injection_INVOICE_DONE_DIRECTORY[]) then
			THROW USER EXCEPTION VALUES('property post_injection_INVOICE_DONE_DIRECTORY not found'); 
		elseif	  not EXISTS(RefSftpEndPointsPolicy.post_injection_INVOICE_ERROR_DIRECTORY[]) then
			THROW USER EXCEPTION VALUES('property post_injection_INVOICE_ERROR_DIRECTORY not found');
		elseif	  not EXISTS(RefSftpEndPointsPolicy.post_injection_INVOICE_SAVE_DIRECTORY[]) then
			THROW USER EXCEPTION VALUES('property post_injection_INVOICE_SAVE_DIRECTORY not found');
		elseif	  not EXISTS(RefSftpEndPointsPolicy.post_injection_INVOICE_INPUT_DIRECTORY[]) then
			THROW USER EXCEPTION VALUES('property post_injection_INVOICE_INPUT_DIRECTORY not found');
		elseif	  not EXISTS(RefSftpEndPointsPolicy.post_injection_INVOICE_LOCAL_DIRECTORY[]) then
			THROW USER EXCEPTION VALUES('property post_injection_INVOICE_LOCAL_DIRECTORY not found');
		end if;
		SET OutputLocalEnvironment.Destination.File.Directory = RefSftpEndPointsPolicy.post_injection_INVOICE_LOCAL_DIRECTORY ;
		SET OutputLocalEnvironment.Destination.File.Remote.ServerDirectory = RefSftpEndPointsPolicy.post_injection_INVOICE_INPUT_DIRECTORY;
		RETURN TRUE;
	END;

END MODULE;

CREATE COMPUTE MODULE PrepareSaveFile
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		SET OutputRoot = InputRoot;
		DECLARE currentTimestamp CHARACTER CAST(CURRENT_TIMESTAMP AS CHARACTER FORMAT 'yyyyMMdd_HHmmss');

		SET OutputLocalEnvironment.Destination.File.Remote.ServerDirectory = RefSftpEndPointsPolicy.post_injection_INVOICE_SAVE_DIRECTORY;
		RETURN TRUE;
	END;

END MODULE;

CREATE COMPUTE MODULE post_injection_tcs_invoice_split_validation
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		DECLARE isSansys BOOLEAN COALESCE(Environment.Variables.IngestionSource, '') = 'SANSYS';
		
		DELETE FIELD Environment.failedInvoices;
		DELETE FIELD Environment.Variables.ValidationFailedJson;
		SET Environment.sapCallFail = NULL;
		SET Environment.invalidFile = FALSE;
		
		IF isSansys THEN
			DELETE FIELD Environment.Variables.TCSLog.context."business-identifiers";
		END IF;
		
		-- ERROR HANDLE will execute the code inside in case of any error
		DECLARE EXIT HANDLER FOR SQLSTATE VALUE ''
        BEGIN
        	SET Environment.invalidFile = TRUE;
        	PROPAGATE TO TERMINAL 'out3' DELETE NONE;
        	IF isSansys THEN
        		PROPAGATE TO TERMINAL 'out5' DELETE NONE;
        	END IF;
            RETURN FALSE ;
        END;
		
		
		IF EXISTS(InputRoot.JSON.Data.invoices.Item[]) = FALSE THEN
			SET Environment.invalidFile = TRUE;
			PROPAGATE TO TERMINAL 'out3' DELETE NONE;
			IF isSansys THEN
				PROPAGATE TO TERMINAL 'out5' DELETE NONE;
			END IF;
			RETURN FALSE;
		END IF;
		
		SET OutputRoot = InputRoot;
		DECLARE inputRef REFERENCE TO InputRoot.JSON.Data.invoices;
		DECLARE I INTEGER 1;
		DECLARE invoiceCount INTEGER;
		DECLARE validationErrorCount INTEGER 0;
		
		-- Get the number of invoices
		SET invoiceCount = CARDINALITY(inputRef.[]);
		DECLARE RefSftpEndPointsPolicy REFERENCE TO Environment.Variables.Policy."finance-integration-policies-endpoints".SftpEndPointsPolicy;
		-- Loop through all invoices
		WHILE I <= invoiceCount DO

			DECLARE jsonDoc REFERENCE TO inputRef.[I];
			DECLARE articleList REFERENCE TO jsonDoc.articles;
			DECLARE debtorRef REFERENCE TO jsonDoc.debtor;
			DECLARE patientRef REFERENCE TO jsonDoc.patient;
			DECLARE interventionRef REFERENCE TO jsonDoc.intervention;
			DECLARE validationFailed BOOLEAN FALSE;
			DECLARE transportNumber CHARACTER;
			
			SET transportNumber = GetReferenceIdentifier(interventionRef.references, 'TRANSPORT_NUMBER');
			IF transportNumber = '' THEN
				SET transportNumber = GetReferenceIdentifier(jsonDoc.references, 'TRANSPORT_NUMBER');
			END IF;
			
			Validation:	LOOP 
				IF FIELDNAME(jsonDoc.id) IS NULL OR
				   FIELDNAME(jsonDoc.date) IS NULL OR
				   FIELDNAME(jsonDoc.total_price) IS NULL OR
				   FIELDNAME(jsonDoc.intervention) IS NULL OR
				   FIELDNAME(jsonDoc.articles) IS NULL OR
				   FIELDNAME(jsonDoc.patient) IS NULL THEN
					SET validationFailed = TRUE;
					LEAVE Validation;
				END IF;

				IF COALESCE(jsonDoc.document_type, '') = '2' AND COALESCE(jsonDoc.invoice_id, '') = ''
				THEN
					SET validationFailed = TRUE;
					LEAVE Validation;
				END IF;

				IF FIELDNAME(interventionRef) IS NOT NULL THEN
					IF FIELDNAME(interventionRef.id) IS NULL OR
					   FIELDNAME(interventionRef.type) IS NULL OR
					   FIELDNAME(interventionRef.organization_id) IS NULL OR
					   FIELDNAME(interventionRef.organization_name) IS NULL OR
					   FIELDNAME(interventionRef.base_id) IS NULL OR
					   FIELDNAME(interventionRef.base_name) IS NULL THEN
						SET validationFailed = TRUE;
						LEAVE Validation;
					END IF;
				END IF;
				
				IF FIELDNAME(patientRef) IS NOT NULL THEN
					IF (NOT isSansys AND FIELDNAME(patientRef.fip_number) IS NULL) OR
					   FIELDNAME(patientRef.category_id) IS NULL OR
					   FIELDNAME(patientRef.category_name) IS NULL THEN
						SET validationFailed = TRUE;
						LEAVE Validation;
					END IF;
				END IF;
				
				IF isSansys AND transportNumber = '' THEN
					SET validationFailed = TRUE;
					LEAVE Validation;
				END IF;
				
				IF FIELDNAME(debtorRef) IS NOT NULL THEN
					IF FIELDNAME(debtorRef.type_id) IS NULL OR
					   FIELDNAME(debtorRef.type_name) IS NULL THEN
						SET validationFailed = TRUE;
						LEAVE Validation;
					END IF;
				END IF;
								
				IF FIELDNAME(articleList) IS NOT NULL THEN
					MOVE articleList FIRSTCHILD;
					WHILE LASTMOVE(articleList) DO
						IF FIELDNAME(articleList.line_1) IS NULL THEN
							SET validationFailed = TRUE;
							LEAVE Validation;
						END IF;
						MOVE articleList NEXTSIBLING;
					END WHILE;
				END IF;
				LEAVE Validation;
			END LOOP Validation;
		
			IF validationFailed THEN
				DELETE FIELD OutputRoot.JSON.Data.invoices.[I - validationErrorCount];
				SET validationErrorCount = validationErrorCount + 1;
				IF EXISTS(Environment.Variables.ValidationFailedJson[]) = FALSE THEN
					CREATE LASTCHILD OF Environment.Variables.ValidationFailedJson DOMAIN('JSON');
					CREATE FIELD Environment.Variables.ValidationFailedJson.JSON.Data.(JSON.Array)invoices;
				END IF;
				SET Environment.Variables.ValidationFailedJson.JSON.Data.invoices.Item[validationErrorCount] = jsonDoc;
			ELSE
				--ADD business-id
				SET Environment.Variables.TCSLog.context."business-identifiers"."INVOICE_ID" = COALESCE(Environment.Variables.TCSLog.context."business-identifiers"."INVOICE_ID", '') 
																								|| '|' || CAST(jsonDoc.id AS CHARACTER);
				IF isSansys THEN
					SET Environment.Variables.TCSLog.context."business-identifiers"."TRANSPORT_NUMBER" = COALESCE(Environment.Variables.TCSLog.context."business-identifiers"."TRANSPORT_NUMBER", '') 
																									|| '|' || transportNumber;
				ELSE
					SET Environment.Variables.TCSLog.context."business-identifiers"."FIP_NUMBER" = COALESCE(Environment.Variables.TCSLog.context."business-identifiers"."FIP_NUMBER", '') 
																								|| '|' || CAST(patientRef.fip_number AS CHARACTER);
				END IF;
				SET Environment.Variables.TCSLog.context."business-identifiers"."INVOICE_REFERENCE" = COALESCE(Environment.Variables.TCSLog.context."business-identifiers"."INVOICE_REFERENCE", '') 
																								|| '|' || CAST(jsonDoc.invoice_id AS CHARACTER);
				SET Environment.Variables.TCSLog.context."business-identifiers"."DOCUMENT_TYPE" = COALESCE(Environment.Variables.TCSLog.context."business-identifiers"."DOCUMENT_TYPE", '') 
																								|| '|' || CAST(jsonDoc.document_type AS CHARACTER);
			END IF;
			
			SET I = I + 1;
		END WHILE;		
		
		-- Checks if all invoices are invalid Send valid messages to routing
		IF EXISTS(OutputRoot.JSON.Data.invoices.Item[1].id[]) = false THEN
			PROPAGATE TO TERMINAL 'out4' DELETE NONE;
		ELSE 
			PROPAGATE TO TERMINAL 'out1' DELETE NONE;
		END IF;
		
		-- If SAP call fails move all invoices identifier to array
		DECLARE currentIdentifier CHARACTER;
		IF Environment.sapCallFail = 'true' THEN
			FOR outInvoice AS OutputRoot.JSON.Data.invoices.Item[] DO
				IF isSansys THEN
					SET currentIdentifier = GetReferenceIdentifier(outInvoice.intervention.references, 'TRANSPORT_NUMBER');
					IF currentIdentifier = '' THEN
						SET currentIdentifier = GetReferenceIdentifier(outInvoice.references, 'TRANSPORT_NUMBER');
					END IF;
				ELSE
					SET currentIdentifier = COALESCE(CAST(outInvoice.patient.fip_number AS CHARACTER), '');
				END IF;
				CREATE LASTCHILD OF Environment.failedInvoices NAME 'identifier' VALUE currentIdentifier;
			END FOR;
		END IF;
		
		-- Deletes unsuccessful invoices from SAP response
		DECLARE sapRejectedInvoicesCount INTEGER CARDINALITY(Environment.failedInvoices.identifier[]);
		DECLARE outInvoicesCount INTEGER CARDINALITY(OutputRoot.JSON.Data.invoices.Item[]);
		DECLARE i INTEGER 1;
		IF sapRejectedInvoicesCount > 0 THEN	
			WHILE i <= outInvoicesCount DO
				DECLARE outInvoice REFERENCE TO OutputRoot.JSON.Data.invoices.Item[i];
				IF isSansys THEN
					SET currentIdentifier = GetReferenceIdentifier(outInvoice.intervention.references, 'TRANSPORT_NUMBER');
					IF currentIdentifier = '' THEN
						SET currentIdentifier = GetReferenceIdentifier(outInvoice.references, 'TRANSPORT_NUMBER');
					END IF;
				ELSE
					SET currentIdentifier = COALESCE(CAST(outInvoice.patient.fip_number AS CHARACTER), '');
				END IF;
				
			 	FOR failedIdentifier AS Environment.failedInvoices.identifier[] DO
					IF currentIdentifier = CAST(failedIdentifier AS CHARACTER) THEN
						IF EXISTS(Environment.Variables.ValidationFailedJson[]) = FALSE THEN
							CREATE LASTCHILD OF Environment.Variables.ValidationFailedJson DOMAIN('JSON');
							CREATE FIELD Environment.Variables.ValidationFailedJson.JSON.Data.(JSON.Array)invoices;
						END IF;
						CREATE LASTCHILD OF Environment.Variables.ValidationFailedJson.JSON.Data.invoices FROM outInvoice;
						DELETE FIELD outInvoice;
						SET i = i - 1;
						SET outInvoicesCount = outInvoicesCount - 1;
					END IF;
				END FOR;
				SET i = i + 1;
			END WHILE;
		END IF;

		
		--Send Report Finish validation
		SET Environment.Variables.ValidInvoicesCount = invoiceCount - validationErrorCount - sapRejectedInvoicesCount;
		SET Environment.Variables.InvalidInvoicesCount = validationErrorCount;
		SET Environment.Variables.SapRejectedInvoicesCount = sapRejectedInvoicesCount;
		SET Environment.Variables.InvoiceCount = invoiceCount;
		PROPAGATE TO TERMINAL 'out' DELETE NONE;
		
		--Write files if Validation error
		IF validationErrorCount > 0 OR sapRejectedInvoicesCount > 0 THEN
			SET OutputRoot = NULL;
			CREATE LASTCHILD OF OutputRoot DOMAIN('JSON');
			CREATE FIELD OutputRoot.JSON.Data.invoices IDENTITY (JSON.Array)invoices;
			SET OutputRoot.JSON.Data.invoices = Environment.Variables.ValidationFailedJson.JSON.Data.invoices;
			DECLARE currentTimestamp CHARACTER CAST(CURRENT_TIMESTAMP AS CHARACTER FORMAT 'yyyyMMdd_HHmmss');
			SET Environment.Variables.TCSLog.context."error-code" = 400;
			SET OutputLocalEnvironment.Destination.File.Name = currentTimestamp || '_validation_failed_invoices.json';
			SET OutputLocalEnvironment.Destination.File.Directory = RefSftpEndPointsPolicy.post_injection_INVOICE_LOCAL_DIRECTORY ;
			SET OutputLocalEnvironment.Destination.File.Remote.ServerDirectory = RefSftpEndPointsPolicy.post_injection_INVOICE_ERROR_DIRECTORY;
			PROPAGATE TO TERMINAL 'out2';
		END IF;
		
		IF isSansys THEN
			PROPAGATE TO TERMINAL 'out5' DELETE NONE;
		END IF;
		RETURN FALSE;
	END;
	
	CREATE FUNCTION GetReferenceIdentifier(IN referencesRef REFERENCE, IN schemeIdentifier CHARACTER) RETURNS CHARACTER
	BEGIN
		DECLARE referencesCount INTEGER CARDINALITY(referencesRef.*[]);
		DECLARE index INTEGER 1;
		DECLARE currentSchemeIdentifier CHARACTER;
		
		WHILE index <= referencesCount DO
			DECLARE referenceEntry REFERENCE TO referencesRef.*[index];
			SET currentSchemeIdentifier = COALESCE(CAST(referenceEntry."scheme-identifier" AS CHARACTER),
												  CAST(referenceEntry.scheme_identifier AS CHARACTER),
												  CAST(referenceEntry.schemeIdentifier AS CHARACTER),
												  '');
			IF UPPER(currentSchemeIdentifier) = UPPER(schemeIdentifier) THEN
				RETURN COALESCE(CAST(referenceEntry.identifier AS CHARACTER),
								CAST(referenceEntry.content AS CHARACTER),
								'');
			END IF;
			SET index = index + 1;
		END WHILE;
		
		RETURN '';
	END;
END MODULE;

CREATE COMPUTE MODULE PrepareOutFiles
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		SET OutputRoot = InputRoot;
		--Write files if Validation error
		DECLARE ValidInvoicesCount INTEGER Environment.Variables.ValidInvoicesCount;
		DECLARE InvalidInvoicesCount INTEGER Environment.Variables.InvalidInvoicesCount;
		DECLARE RejectedInvoicesCount INTEGER Environment.Variables.SapRejectedInvoicesCount;
		DECLARE TotalInvoicesCount INTEGER Environment.Variables.InvoiceCount;
		DECLARE RefSftpEndPointsPolicy REFERENCE TO Environment.Variables.Policy."finance-integration-policies-endpoints".SftpEndPointsPolicy;
		
		IF ValidInvoicesCount > 0 THEN
			DECLARE currentTimestamp CHARACTER CAST(CURRENT_TIMESTAMP AS CHARACTER FORMAT 'yyyyMMdd_HHmmss');
			SET OutputLocalEnvironment.Destination.File.Name = currentTimestamp ||'_invoices.json';
			SET OutputLocalEnvironment.Destination.File.Directory = RefSftpEndPointsPolicy.post_injection_INVOICE_LOCAL_DIRECTORY ;
			SET OutputLocalEnvironment.Destination.File.Remote.ServerDirectory =RefSftpEndPointsPolicy.post_injection_INVOICE_DONE_DIRECTORY;
			PROPAGATE TO TERMINAL 'out';
		END IF;
		
		--Generate Report
		DECLARE EventMessage, ingestionStatus CHARACTER;
		SET Environment.Variables.TCSLog.context."business-identifiers" = NULL;
		IF TotalInvoicesCount = ValidInvoicesCount THEN
			SET EventMessage = 'SUP: One File processed successully';
			SET ingestionStatus = 'Success';	
			SET Environment.Variables.TCSLog.context.severity = 'INFO';
		ELSEIF TotalInvoicesCount = InvalidInvoicesCount + RejectedInvoicesCount THEN
			SET EventMessage = 'SUP: One File failed to be processed entirely';
			SET ingestionStatus = 'Failure';	
			SET Environment.Variables.TCSLog.context.severity = 'ERR';
		ELSE 
			SET EventMessage = 'SUP: One File processed with errors';
			SET ingestionStatus = 'Partial Success';
			SET Environment.Variables.TCSLog.context.severity = 'WARN';
		END IF;	
			
		SET Environment.Variables.TCSLog.context."business-identifiers"."INGESTION_STATUS" = ingestionStatus;
		SET Environment.Variables.TCSLog.context."business-identifiers"."TOTAL_PROCESSED_INVOICES" = TotalInvoicesCount;
		SET Environment.Variables.TCSLog.context."business-identifiers"."TOTAL_SUCCESFULLY_PROCESSED_INVOICES" = ValidInvoicesCount;
		SET Environment.Variables.TCSLog.context."business-identifiers"."TOTAL_WRONG_INPUT_FORMAT_ERRORS" = InvalidInvoicesCount;
		SET Environment.Variables.TCSLog.context."business-identifiers"."TOTAL_REJECTION_ERRORS" = RejectedInvoicesCount;
		SET Environment.Variables.EventMessage = EventMessage;
		PROPAGATE TO TERMINAL 'out2';
		RETURN FALSE;
	END;
END MODULE;

CREATE COMPUTE MODULE emptyOrMissingFileError
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		SET OutputLocalEnvironment = InputLocalEnvironment;

		IF COALESCE(Environment.Variables.BatchMode, FALSE) = TRUE
		   AND COALESCE(Environment.Variables.IngestionSource, '') = 'SANSYS'
		   AND InputLocalEnvironment.File.Read.NoMatchReason = 'NoFile'
		   AND COALESCE(Environment.Variables.BatchProcessedFileCount, 0) > 0 THEN			
			IF COALESCE(Environment.Variables.BatchLoopLimitReached, FALSE) THEN
				SET Environment.Variables.TCSLog.context.severity = 'WARN';
			ELSE
				SET Environment.Variables.TCSLog.context.severity = 'INFO';
			END IF;

			SET OutputRoot.JSON.Data.meta.submissionId = COALESCE(NULLIF(Environment.Variables."X-Global-Transaction-Id", ''),COALESCE(NULLIF(Environment.Variables."X-Correlation-Id", ''),COALESCE(NULLIF(Environment.Variables.trackID, ''), UUIDASCHAR)));
			SET OutputRoot.JSON.Data.meta.timestamp = CAST(CURRENT_TIMESTAMP AS CHARACTER FORMAT 'yyyy-MM-dd''T''HH:mm:ss');
			SET OutputRoot.JSON.Data.data.status = 'COMPLETED';
			SET OutputRoot.JSON.Data.data.receivedCount = COALESCE(Environment.Variables.BatchProcessedFileCount, 0);
			IF COALESCE(Environment.Variables.BatchLoopLimitReached, FALSE) THEN
				SET OutputRoot.JSON.Data.data.message = 'Batch file ingestion stopped after reaching the maximum configured iteration count.';
			ELSE
				SET OutputRoot.JSON.Data.data.message = 'Batch file ingestion completed.';
			END IF;

			SET OutputLocalEnvironment.Destination.HTTP.ReplyStatusCode = 200;

			RETURN TRUE;
		END IF;
		
		IF InputLocalEnvironment.File.Read.NoMatchReason = 'NoFile' THEN			
			SET Environment.Variables.TCSLog.context.severity = 'WARN';

			SET OutputRoot.JSON.Data.meta.submissionId = COALESCE(NULLIF(Environment.Variables."X-Global-Transaction-Id", ''),COALESCE(NULLIF(Environment.Variables."X-Correlation-Id", ''),COALESCE(NULLIF(Environment.Variables.trackID, ''), UUIDASCHAR)));
			SET OutputRoot.JSON.Data.meta.timestamp = CAST(CURRENT_TIMESTAMP AS CHARACTER FORMAT 'yyyy-MM-dd''T''HH:mm:ss');
			SET OutputRoot.JSON.Data.data.status = 'NO_FILE';
			SET OutputRoot.JSON.Data.data.receivedCount = 0;
			SET OutputRoot.JSON.Data.data.message = '';

			SET OutputLocalEnvironment.Destination.HTTP.ReplyStatusCode = 200;

			RETURN TRUE;
		END IF;

		IF Environment.invalidFile = TRUE THEN			
			SET Environment.Variables.TCSLog.context.severity = 'ERR';

			SET OutputLocalEnvironment.Destination.HTTP.ReplyStatusCode = 422;

			DECLARE output REFERENCE TO OutputRoot.JSON.Data;
			
			SET OutputRoot.JSON.Data.httpCode = '422';
			SET OutputRoot.JSON.Data.httpMessage = 'Unprocessable Entity';
			SET OutputRoot.JSON.Data.moreInformation = 'Invalid file format or content. The payload could not be parsed.';
			
			RETURN TRUE;
		END IF;
	RETURN TRUE;
	END;
END MODULE;

	CREATE PROCEDURE AddInvoiceNote(IN invoiceNoteRef REFERENCE, IN notePosition INTEGER, IN subjectCode CHARACTER, IN noteText CHARACTER)
	BEGIN
		SET invoiceNoteRef.Item[notePosition]."invoiceNoteSubjectCode".content = subjectCode;
		SET invoiceNoteRef.Item[notePosition]."invoiceNoteText".content = COALESCE(noteText, '');
	END;
	
	CREATE FUNCTION GetReferenceIdentifier(IN referencesRef REFERENCE, IN schemeIdentifier CHARACTER) RETURNS CHARACTER
	BEGIN
		DECLARE referencesCount INTEGER CARDINALITY(referencesRef.*[]);
		DECLARE index INTEGER 1;
		DECLARE currentSchemeIdentifier CHARACTER;
		
		WHILE index <= referencesCount DO
			DECLARE referenceEntry REFERENCE TO referencesRef.*[index];
			SET currentSchemeIdentifier = COALESCE(CAST(referenceEntry."scheme-identifier" AS CHARACTER),
												  CAST(referenceEntry.scheme_identifier AS CHARACTER),
												  CAST(referenceEntry.schemeIdentifier AS CHARACTER),
												  '');
			IF UPPER(currentSchemeIdentifier) = UPPER(schemeIdentifier) THEN
				RETURN COALESCE(CAST(referenceEntry.identifier AS CHARACTER),
								CAST(referenceEntry.content AS CHARACTER),
								'');
			END IF;
			SET index = index + 1;
		END WHILE;
		
		RETURN '';
	END;
	
	CREATE FUNCTION ResolveBukrs(IN baseId CHARACTER) RETURNS CHARACTER
	BEGIN
		DECLARE propertyName CHARACTER;
		DECLARE bukrs CHARACTER;
		
		SET propertyName = 'sansysmit_' || LOWER(COALESCE(baseId, '')) || '_bukrs';
		SET bukrs = Shlib_PoliciesReader.getPolicyPropertyWithOutError('{finance-integration-policies-endpoints}:BaseIdentityPolicy', propertyName);
		
		IF bukrs = 'POLICY_NOT_FOUND' OR bukrs = 'PROPERTY_NOT_FOUND' OR COALESCE(TRIM(bukrs), '') = '' THEN
			SET propertyName = 'sansysmit_' || COALESCE(baseId, '') || '_bukrs';
			SET bukrs = Shlib_PoliciesReader.getPolicyPropertyWithOutError('{finance-integration-policies-endpoints}:BaseIdentityPolicy', propertyName);
		END IF;
		
		IF bukrs = 'POLICY_NOT_FOUND' OR bukrs = 'PROPERTY_NOT_FOUND' OR COALESCE(TRIM(bukrs), '') = '' THEN
			SET propertyName = 'SANSYSMIT_' || COALESCE(baseId, '') || '_bukrs';
			SET bukrs = Shlib_PoliciesReader.getPolicyPropertyWithOutError('{finance-integration-policies-endpoints}:BaseIdentityPolicy', propertyName);
		END IF;
		
		RETURN bukrs;
	END;
	
	CREATE FUNCTION StripHyphen(IN inputValue CHARACTER) RETURNS CHARACTER
	BEGIN
		RETURN TRANSLATE(COALESCE(inputValue, ''), '-', '');
	END;

	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		SET OutputRoot = InputRoot;
		SET OutputLocalEnvironment = InputLocalEnvironment;

---COMMENT---
Summary:
- Reworked the SANSYS injection flow into a batch-oriented trigger flow.
- Added a new Sansys EOM v2 camelCase mapper and Sansys batch control ESQL.
- Added source-trigger-based validation branching so SANSYS no longer requires `fip_number` and now requires `TRANSPORT_NUMBER`.
- Repointed SANSYS outbound URL resolution to the v2 receivable-documents endpoint property.
- Added the new Ace endpoint property and introduced a new BaseIdentityPolicy scaffold.
- Updated the OpenAPI response schema to the current `meta/data` response style.

What changed and why:
1. `postSansys_files_injection_trigger.subflow`
   - Added Sansys batch initialization.
   - Added file-count tracking and a queue-free loop back to re-read the folder until `noMatch`.
   - Kept `ProcessFile.noMatch -> emptyOrMissingFileError` for empty-folder / end-of-batch response.
   - Switched the SANSYS request mapping node to a new v2 mapper.
   - Updated the downstream monitored call to `finance-accounting-integration-invoices-v2` / `POST /receivable-documents`.
   - Prevented per-file HTTP replies; only the final batch completion path now replies.

2. `PrepareAndProcessFile.esql`
   - Added trigger-based URL selection for SANSYS vs existing Attrib flows.
   - Added SANSYS validation rules:
     - no `fip_number` requirement
     - `category_id` / `category_name` still required
     - `TRANSPORT_NUMBER` required from references
   - Replaced SANSYS logging business identifier from `FIP_NUMBER` to `TRANSPORT_NUMBER`.
   - Reset per-file transient Environment content so batch iterations do not leak state.
   - Added a new internal terminal `out5` used only by the SANSYS batch loop.
   - Adjusted SAP-failure invoice matching logic so SANSYS uses transport-number-based identifiers instead of `fip_number`.

3. `sansys/PrepareReceivableDocumentRequest.esql`
   - New mapper producing camelCase EOM v2 payloads.
   - Hardcoded `processControl.specificationIdentifier = AMBULANCE_TRANSPORT_BILLING`.
   - Emits:
     - `invoiceNumber[INVOICE_ID]`
     - `invoiceNumber[TRANSPORT_NUMBER]`
     - `seller.sellerIdentifier[COMPANY_CODE]`
     - `buyer.buyerIdentifier[SANSYS]`
     - updated invoiceNote codes including `INVOICE_REFERENCE`, `INTERVENTION_CASE_NUMBER`, `PATIENT_ID`
   - Keeps local timestamp strings unchanged.
   - Uses policy-driven `bukrs` lookup from `base_id`.
   - On missing mapping sets `Environment.Variables.ErrorRefApp = ACEFONC0023` and stops the request branch loudly.

4. `sansys/BatchControl.esql`
   - New helper modules for:
     - initializing batch context
     - counting processed files
     - stopping the loop after a configurable iteration cap

5. `emptyOrMissingFileError.esql`
   - Added a SANSYS batch completion branch:
     - zero files -> existing `NO_FILE`
     - one or more files -> final batch response with `meta/data`

6. Policy project changes
   - `AceEndPointsPolicy.policyxml`: added `path-finance-accounting-integration-invoices-v2-receivable-documents`
   - `BaseIdentityPolicy.policyxml`: added a new user-defined policy file for `producer_app + base_id -> bukrs`

7. OpenAPI file
   - Updated `IngestionResponse` from the old `status/summary/results` structure to the current `meta/data` structure.

Assumptions / limitations / uncertainties:
- I did not change the ASV / SARGE / SARVD injection flows or their mappers.
- I kept the existing SANSYS SFTP identity `SANSYSAceSftpIdentity` because it is already a dedicated non-ASV identity in the current flow.
- The full base-identity lookup table was not provided in the repository snapshot. I only added the policy mechanism and one explicit example mapping from the request:
  - `sansysmit_c7b01f98-70b1-457c-bd1a-95bf1ffeb548_bukrs = 1070`
  The remaining mappings must be completed from the external identity map before rollout.
- The exact final batch HTTP status semantics were not specified for non-empty folders. I returned a generic `200` batch completion response in `meta/data` form.
- The final batch response `receivedCount` is implemented as processed file count, not invoice count, because the request focused on folder-drain behavior and did not define a batch response aggregation contract.
- The exact input reference structure was not fully shown; I implemented reference lookup tolerant to:
  - `scheme-identifier`
  - `scheme_identifier`
