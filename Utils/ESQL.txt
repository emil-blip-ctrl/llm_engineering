---------------------------------------COPYING MESSAGES-------------------------------------
The following single statement copies the iterations of the input fields to the output fields, and deletes any surplus fields in the output message.

SET OutputRoot.XMLNS.Output_top.Outfield1.[] = InputRoot.XMLNS.Field_top.field1[];

The following example shows how you can rename the elements when you copy them into the output tree.
 This statement does not copy across the source element name, therefore each field1 element becomes a Target element.

SET OutputRoot.XMLNS.Output_top.Outfield1.Target[] = 
    (SELECT I FROM InputRoot.XMLNS.Field_top.field1[] AS I );

The next example shows a different way to do the same operation; it produces the same end result.

SET OutputRoot.XMLNS.Output_top.Outfield2.Target[] 
              = InputRoot.XMLNS.Field_top.field1[];  

The following example copies across the source element name. Each field1 element is retained as a field1 element under the Target element.

SET OutputRoot.XMLNS.Output_top.Outfield3.Target.[] 
              = InputRoot.XMLNS.Field_top.field1[];  

This example is an alternative way to achieve the same result, with field1 elements created under the Target element.

SET OutputRoot.XMLNS.Output_top.Outfield4.Target.*[] 
              = InputRoot.XMLNS.Field_top.field1[];

	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		
		SET Environment.Variables.Error = true;
		
		CALL CopyMessageHeaders();

		SET OutputRoot.HTTPReplyHeader."X-Correlation-ID" = Environment.Variables.XCorrelationID;

		SET Environment.Variables.HttpReturnCode = Environment.Variables.statusCode;
		SET OutputLocalEnvironment.Destination.HTTP.ReplyStatusCode = Environment.Variables.HttpReturnCode;
		SET OutputRoot.JSON.Data.timestamp = CAST(CURRENT_TIMESTAMP AS CHARACTER FORMAT 'I');
		SET OutputRoot.JSON.Data.httpCode = '500';
		SET OutputRoot.JSON.Data.httpMessage = 'ACE : Internal Server Error';
		SET OutputRoot.JSON.Data.contexteCode = '';
		SET OutputRoot.JSON.Data.id = Environment.Variables."X-Global-Transaction-Id";
		IF EXISTS(InputExceptionList.RecoverableException[]) THEN
			SET OutputRoot.JSON.Data.moreInformation = 'Error Calling Salesforce: ' || GetDeepestInsertTextsSafe(InputExceptionList);
		ELSE
			SET OutputRoot.JSON.Data.moreInformation = 'Error Calling Salesforce: | Message: ' || COALESCE(InputRoot.JSON.Data.Item[1].message, '') || 
													   '| Error code: ' || COALESCE(InputRoot.JSON.Data.Item[1].errorCode, '');
		END IF;
		
		SET OutputLocalEnvironment.Error = true;
		SET OutputLocalEnvironment.Destination.HTTP.ReplyStatusCode = '500';
		SET OutputLocalEnvironment.Destination.HTTP.RequestIdentifier= Environment.Variables.RequestIdentifier;
		SET OutputLocalEnvironment.Environment= Environment;

		SET Environment.Variables.TCSLog.context.severity = 'ERROR';
		SET Environment.Variables.TCSLog.context."source-system" = 'CMP05541';
		SET Environment.Variables.TCSLog.context."target-system" = 'CMP05541';
		SET Environment.Variables.TCSLog.context."caller-code" = Environment.Variables.In.XCallerCode;
		SET Environment.Variables.TCSLog.context."target-url" = Environment.Variables.In.targeturl;
		SET Environment.Variables.TCSLog.context."error-message" = OutputRoot.JSON.Data.statusMessage || '. ' || OutputRoot.JSON.Data.moreInformation;
		SET Environment.Variables.TCSLog.context."error-code" = OutputRoot.JSON.Data.statusCode;	
		
		RETURN TRUE;
	END;
	
	CREATE FUNCTION GetDeepestInsertTextsSafe(exceptionTree REFERENCE) RETURNS CHARACTER
	BEGIN
		DECLARE result CHARACTER '';
		DECLARE recovarableException REFERENCE TO exceptionTree.RecoverableException;

		IF LASTMOVE(recovarableException) THEN
			RETURN GetDeepestInsertTextsSafe(recovarableException);
		END IF;
		
		SET result = exceptionTree.Text || ': ';
		DECLARE insertElement REFERENCE TO exceptionTree.Insert[1];

		WHILE LASTMOVE(insertElement) DO
			IF FIELDNAME(insertElement.Text) IS NOT NULL THEN
				SET result = result || '|' || insertElement.Text;
			END IF;
			MOVE insertElement NEXTSIBLING;
		END WHILE;
		
		RETURN result || '|';
	END;
	
	CREATE PROCEDURE CopyMessageHeaders() BEGIN
		DECLARE I INTEGER 1;
		DECLARE J INTEGER;
		SET J = CARDINALITY(InputRoot.*[]);
		WHILE I < J DO
			SET OutputRoot.*[I] = InputRoot.*[I];
			SET I = I + 1;
		END WHILE;
	END;
	
END MODULE;
