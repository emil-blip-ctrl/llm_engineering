info:
  description: >
    API for Enterprise Content Management (ECM) document event notifications.

    This API allows external systems (e.g., OnBase) to notify document-related events
    with associated metadata and keywords.

    ### 🔑 Key Features

    - 📄 Receive document metadata (type, dates, creator, etc.)

    - 🏷️ Handle structured keywords and keyword groups

    - 🔗 Support for multi-index keyword groups

    - ⚡ Real-time document event notification


    ### 📦 Payload Structure

    - **properties**
      - Document identification and metadata
      - Includes document type, creation info, and lifecycle dates

    - **keywords**
      - List of keywords or keyword groups
      - Supports nested structures for complex classification

  /notify:
    post:
      summary: postNotification

        description: OnBase document notification payload
        required: true

          application/json:
            schema:
              $ref: '#/components/schemas/OnbaseDocument'

        '200':

        '500':
          description: Unexpected Error
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/Error'
components:
  schemas:

    Error:
      type: object
      required:
        - httpCode
        - httpMessage
        - moreInformation

        httpCode:
          type: string
        httpMessage:
          type: string
        moreInformation:
          type: string

    OnbaseDocument:
      type: object
      required:
        - properties
        - keywords
      properties:
        properties:
          $ref: '#/components/schemas/OnbaseProperties'
        keywords:
          type: array
          items:
            $ref: '#/components/schemas/KeywordBase'
      example:

            keyword-datatype: Alphanumeric

    OnbaseProperties:
      type: object
      required:
        - doc-id
        - document-type-id
      properties:
        doc-id:
          type: integer
          description: OnBase document identifier
        created-by:
          type: string
          description: samAccountName or email used during document creation
        document-type-id:
          type: integer
        document-type-name:
          type: string
        document-date:
          type: string
          format: date-time
        document-archival-date:
          type: string
          format: date-time
        document-group-id:
          type: integer
        document-group-name:
          type: string

    KeywordBase:
      type: object
      required:
        - "type"
      properties:
        "type":
          type: string
      oneOf:
        - $ref: '#/components/schemas/Keyword'
        - $ref: '#/components/schemas/KeywordGroup'

          type: string

          oneOf:
